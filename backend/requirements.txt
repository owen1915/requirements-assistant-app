# Dependencies live in the repo-root pyproject.toml, so nothing can drift onto
# a different version of the same library. This file is a thin shim kept only
# so an older instruction still works:
#
#     pip install -r backend/requirements.txt     # equivalent to the line below
#     pip install -e .[app]                       # what it actually does
#
# RUN IT FROM THE REPOSITORY ROOT. pip resolves a relative path in a
# requirements file against the current working directory, NOT against the
# directory holding the file — so `-e .` below means "wherever you invoked pip",
# and running this from inside backend/ will fail to find pyproject.toml.
-e .[app]
